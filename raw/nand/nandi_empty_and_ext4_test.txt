cd ~/bb/nand

for img in nand*.img; do
  base="${img%.img}"

  echo "[*] Deep analyzing $img"

  # Filesystem / signature detection
  blkid "$img" > ~/bb/analysis/fsdetect/${base}.blkid.txt 2>&1
  fdisk -l "$img" > ~/bb/analysis/fsdetect/${base}.fdisk.txt 2>&1

  # Entropy (detect compressed/encrypted blobs)
  binwalk -E "$img" > ~/bb/analysis/entropy/${base}.entropy.txt

  # Full signature scan
  binwalk -A "$img" > ~/bb/analysis/binwalk/${base}.signatures.txt
done
